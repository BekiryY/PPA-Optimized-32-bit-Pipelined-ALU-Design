exxample:

module top_module (
    input logic clk,
    input logic low_power, // 1=Use Slow, 0=Use Fast
    // ... other I/O
    output logic [63:0] result
);
    // 1. Instantiate both multipliers
    fast_mult_7stage fast_inst (.clk(clk), ...);
    slow_mult_comb slow_inst ( ...);

    // 2. Select the output
    assign result = low_power ? slow_inst.out : fast_inst.out;
endmodule




# When low_power is 0 (Fast Path is ON), the slow path is not the critical path.
set_case_analysis 0 [get_ports low_power]
# Now, Genus will find the worst-case path through the 7-stage multiplier.
# It MUST meet the fast clock period (e.g., 0.5 ns).


# Define the slow path (Input to Output of the combinational multiplier)
set SLOW_INPUTS [get_pins slow_inst/*/A]
set SLOW_OUTPUTS [get_pins slow_inst/*/out]

# When low_power is 1 (Slow Path is ON), we must constrain the timing, but loosely.
# We tell Genus that the slow path must meet a very long period (e.g., 5.0 ns).
set_case_analysis 1 [get_ports low_power]
set_max_delay -from $SLOW_INPUTS -to $SLOW_OUTPUTS 5.0
